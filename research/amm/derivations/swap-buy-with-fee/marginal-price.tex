\subsubsection{Equation~\ref{eqn:swap-buy-with-fee-marginal-price} derivation}%
\label{sssec:equation-eqn-swap-buy-with-fee-marginal-price-derivation}

Consider Equation~\ref{eqn:swap-buy-with-fee-taker-execution-price}.

\begin{equation}
  p_t = \frac{q_i + q_{in}}{(1 - f) \cdot b_i} \nonumber
\end{equation}

Take the limit of $p_t$ as $q_{in}$ approaches zero.

\begin{equation}
  p_m = \lim_{q_{in} \to 0} p_t  = \frac{q_i}{(1 - f) \cdot b_i} \nonumber
\end{equation}
