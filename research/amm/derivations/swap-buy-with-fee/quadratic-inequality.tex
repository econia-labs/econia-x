\subsubsection{Equation~\ref{eqn:swap-buy-with-fee-quadratic-inequality} derivation}%
\label{sssec:equation-eqn-swap-buy-with-fee-quadratic-inequality-derivation}

Assume $p_s > p_m$.

\begin{equation*}
	p_s > p_m
\end{equation*}

Substitute equation \ref{eqn:swap-buy-with-fee-marginal-price}.

\begin{equation*}
	p_s > \frac{q_i}{f^{\dagger} \cdot b_i}
\end{equation*}

Multiply both sides by $f^{\dagger} \cdot b_i$, which is nonnegative.

\begin{equation*}
  p_s \cdot f^{\dagger} \cdot b_i > q_i
\end{equation*}

Substitute $Q = (f^{\dagger} \cdot b_i \cdot p_s)$ from
Equation~\ref{eqn:swap-buy-with-fee-slippage-input}.

\begin{equation*}
  Q > q_i
\end{equation*}

Subtract $Q$ from both sides.

\begin{equation*}
  0 > q_i - Q
\end{equation*}

Rearrange.

\begin{equation*}
  q_i - Q < 0
\end{equation*}

Multiply both sides by $q_i$.

\begin{equation*}
  q_i^2 -  q_i \cdot Q < 0
\end{equation*}

Substitute $C$ from Equation~\ref{eqn:swap-buy-with-fee-slippage-input}.

\begin{equation*}
  C < 0
\end{equation*}

Substitute $F$ from Equation~\ref{eqn:swap-buy-with-fee-slippage-input}, with $q_s = 0$.

\begin{equation*}
  F(0) < 0
\end{equation*}
