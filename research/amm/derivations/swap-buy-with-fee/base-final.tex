\subsubsection{Equation~\ref{eqn:swap-buy-with-fee-base-final} derivation}%
\label{sssec:equation-eqn-swap-buy-with-fee-base-final-derivation}

Define $b_f$ as the difference of $b_i$ and $b_{out}$.

\begin{equation}
	b_f = b_i - b_{out} \nonumber
\end{equation}

Substitute Equation~\ref{eqn:swap-buy-with-fee-base-out}.

\begin{equation}
	b_f = b_i - (1 - f) \cdot \frac{b_i \cdot q_{in}}{q_i + q_{in}} \nonumber
\end{equation}

Multiply both sides by $(q_i + q_{in})$.

\begin{equation}
	b_f \cdot (q_i + q_{in}) =
	b_i \cdot (q_i + q_{in}) - (1 - f) \cdot b_i \cdot q_{in} \nonumber
\end{equation}

Expand the right side.

\begin{equation}
	b_f \cdot (q_i + q_{in}) = b_i \cdot q_i + b_i \cdot q_{in} - b_i \cdot q_{in} +
	f \cdot b_i \cdot q_{in} \nonumber
\end{equation}

Cancel the $b_i \cdot q_{in}$ terms on the right side.

\begin{equation}
	b_f \cdot (q_i + q_{in}) = b_i \cdot q_i + f \cdot b_i \cdot q_{in} \nonumber
\end{equation}

Factor out $b_i$ on the right side.

\begin{equation}
	b_f \cdot (q_i + q_{in}) = b_i \cdot (q_i + f \cdot q_{in}) \nonumber
\end{equation}

Divide both sides by $(q_i + q_{in})$.

\begin{equation}
	b_f = b_i \cdot \frac{q_i + f \cdot q_{in}}{q_i + q_{in}} \nonumber
\end{equation}
