\subsubsection{Equation~\ref{eqn:swap-buy-with-fee-taker-execution-price} derivation}%
\label{sssec:equation-eqn-swap-buy-with-fee-taker-execution-price-derivation}

Consider equation Equation~\ref{eqn:swap-taker-execution-price}.

\begin{equation}
  p_t = \frac{\Delta q_t}{\Delta b_t} \nonumber
\end{equation}

Substitute values from Figure~\ref{fig:swap-buy-with-fee}.

\begin{equation}
  p_t = \frac{q_{in}}{b_{out}} \nonumber
\end{equation}

Substitute Equation~\ref{eqn:swap-buy-with-fee-base-out}.
\begin{equation}
  p_t = \frac{q_{in}}{\left((1 - f) \cdot \dfrac{b_i \cdot q_{in}}{q_i + q_{in}}\right)}
  \nonumber
\end{equation}

Multiply the numerator and denominator by $(q_i + q_{in})$.

\begin{equation}
  p_t = \frac{q_{in} \cdot (q_i + q_{in})}{(1 - f) \cdot b_i \cdot q_{in}} \nonumber
\end{equation}

Cancel $q_{in}$.

\begin{align}
  p_t & = \frac{q_i + q_{in}}{(1 - f) \cdot b_i} \nonumber
\end{align}
