\subsubsection{Equation~\ref{eqn:swap-buy-with-fee-base-out} derivation}%
\label{sssec:equation-eqn-swap-buy-with-fee-base-out-derivation}

Define $b_{out}$ in terms of $b_v$ and $f$.

\begin{equation*}
	b_{out} = b_v - f \cdot b_v
\end{equation*}

Group $b_v$ coefficients.

\begin{equation*}
	b_{out} = (1 - f) \cdot b_v
\end{equation*}

Substitute Equation~\ref{eqn:fee-remainder}.

\begin{equation*}
	b_{out} = f^{\dagger} \cdot b_v
\end{equation*}
