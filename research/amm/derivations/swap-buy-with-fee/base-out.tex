\subsubsection{Equation~\ref{eqn:swap-buy-with-fee-base-out} derivation}%
\label{sssec:equation-eqn-swap-buy-with-fee-out-derivation}

Define $b_{out}$ in terms of $b_v$ and $f$.

\begin{equation}
	b_{out} = b_v - f \cdot b_v \nonumber
\end{equation}

Group $b_v$ coefficients.

\begin{equation}
	b_{out} = (1 - f) \cdot b_v \nonumber
\end{equation}

Substitute Equation~\ref{eqn:swap-buy-with-fee-base-volume}.

\begin{equation}
	b_{out} = (1 - f) \cdot \frac{b_i \cdot q_{in}}{q_i + q_{in}} \nonumber
\end{equation}
