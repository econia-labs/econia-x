\subsubsection{Equation~\ref{eqn:swap-buy-with-fee-convergence} derivation}%
\label{sssec:equation-eqn-swap-buy-with-fee-convergence-derivation}

Define a Newton-Raphson update step.

\begin{equation*}
  q_1 = q_0 -\frac{F(q_0)}{F'(q_0)}
\end{equation*}

Substitute Equations~\ref{eqn:swap-buy-with-fee-slippage-input} and
\ref{eqn:swap-buy-with-fee-quadratic-prime}.

\begin{equation*}
  q_1 = q_0 -\frac{q_0^2 + B \cdot q_0 + C}{2 \cdot q_0 + B}
\end{equation*}

Multiply the standalone $q_0$ by $\dfrac{2 \cdot q_0 + B}{2 \cdot q_0 + B}$.

\begin{equation*}
  q_1 = \frac{q_0 \cdot (2 \cdot q_0 + B)}{2 \cdot q_0 + B} -
  \frac{q_0^2 + B \cdot q_0 + C}{2 \cdot q_0 + B}
\end{equation*}

Combine the fractions.

\begin{equation*}
  q_1 = \frac{q_0 \cdot (2 \cdot q_0 + B) - (q_0^2 + B \cdot q_0 + C)}{2 \cdot q_0 + B}
\end{equation*}

Expand the numerator.

\begin{equation*}
  q_1 = \frac{2 \cdot q_0^2 + B \cdot q_0 - q_0^2 - B \cdot q_0 - C}{2 \cdot q_0 + B}
\end{equation*}

Cancel terms as appropriate.

\begin{equation*}
  q_1 = \frac{q_0^2 - C}{2 \cdot q_0 + B}
\end{equation*}

Substitute $B$ and $C$ from Equations~\ref{eqn:swap-buy-with-fee-slippage-input}, with
$q_0 = q_i$.

\begin{equation*}
  q_1 = \frac{q_0^2 - (q_0^2 - q_0 \cdot Q)}{2 \cdot q_0 + (2 \cdot q_i - f \cdot Q)}
\end{equation*}

Expand the numerator and denominator.

\begin{equation*}
  q_1 = \frac{q_0^2 - q_0^2 + q_0 \cdot Q}{2 \cdot q_0 + 2 \cdot q_i - f \cdot Q}
\end{equation*}

Cancel terms as appropriate.

\begin{equation*}
  q_1 = \frac{q_0 \cdot Q}{2 \cdot q_0 + 2 \cdot q_i - f \cdot Q}
\end{equation*}

Note that $F(q_s)$ is a quadratic polynomial in $q_s$ and thus converges to the root in
a single iteration. Moreover, $q_s$ is only required in practice if the maximum
possible quote input amount $q_{max}$ yields a slippage price $p_s$ that is
greater than the best ask price. If this condition is met, then $q_{max}$ is used
as the initial guess for the update step, which necessarily converges to the positive
root since $q_s < q_{max}$ and the function is concave up.

\begin{equation*}
  q_s = \frac{q_{max} \cdot Q}{2 \cdot q_{max} + 2 \cdot q_i - f \cdot Q}
\end{equation*}
