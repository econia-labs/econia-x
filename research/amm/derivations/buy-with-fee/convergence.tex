\subsubsection{Equation~\ref{eqn:buy-with-fee-convergence} derivation}%
\label{sssec:equation-eqn-buy-with-fee-convergence-derivation}

Define a Newton-Raphson update step.

\begin{equation*}
	q_{s, n + 1} = q_{s, n} -\frac{F(q_{s, n})}{F'(q_{s, n})}
\end{equation*}

Substitute $F(q_{s, n})$ from Equation~\ref{eqn:buy-with-fee-polynomial}.

\begin{equation*}
	q_{s, n + 1} = q_{s, n} -\frac{q_{s, n}^2 + B \cdot q_{s, n} + C}{F'(q_{s, n})}
\end{equation*}

Substitute $F'(q_{s, n})$ in the denominator.

\begin{equation*}
	q_{s, n + 1} =
	q_{s, n} -\frac{q_{s, n}^2 + B \cdot q_{s, n} + C}{2 \cdot q_{s, n} + B}
\end{equation*}

Multiply the standalone $q_{s, n}$ term by
$\left(\dfrac{2 \cdot q_{s, n} + B}{2 \cdot q_{s, n} + B}\right)$.

\begin{equation*}
	q_{s, n + 1} = \frac{q_{s, n} \cdot (2 \cdot q_{s, n} + B)}{2 \cdot q_{s, n} + B} -
	\frac{q_{s, n}^2 + B \cdot q_{s, n} + C}{2 \cdot q_{s, n} + B}
\end{equation*}

Combine the fractions.

\begin{equation*}
	q_{s, n + 1} =
	\frac{q_{s, n} \cdot (2 \cdot q_{s, n} + B) - (q_{s, n}^2 + B \cdot q_{s, n} + C)}
	{2 \cdot q_{s, n} + B}
\end{equation*}

Expand the numerator.

\begin{equation*}
	q_{s, n + 1} =
	\frac{2 \cdot q_{s, n}^2 + B \cdot q_{s, n} - q_{s, n}^2 - B \cdot q_{s, n} - C}
	{2 \cdot q_{s, n} + B}
\end{equation*}

Cancel terms as appropriate.

\begin{equation*}
	q_{s, n + 1} = \frac{q_{s, n}^2 - C}{2 \cdot q_{s, n} + B}
\end{equation*}

Substitute $B$ and $C$ from Equations~\ref{eqn:buy-with-fee-polynomial}.

\begin{equation*}
	q_{s, n + 1} = \frac{q_{s, n}^2 - (q_i^2 - q_i \cdot t)}
	{2 \cdot q_{s, n} + (2 \cdot q_i - f \cdot t)}
\end{equation*}

Expand.

\begin{equation*}
	q_{s, n + 1} =
	\frac{q_{s, n}^2 - q_i^2 + q_i \cdot t}{2 \cdot q_{s, n} + 2 \cdot q_i - f \cdot t}
\end{equation*}

Subtract at the end to prevent intermediate numerical underflow.

\begin{equation*}
	q_{s, n + 1} =
	\frac{q_{s, n}^2 + q_i \cdot t - q_i^2}{2 \cdot q_{s, n} + 2 \cdot q_i - f \cdot t}
\end{equation*}
