\subsubsection{Equation~\ref{eqn:buy-with-fee-polynomial-inequality} derivation}%
\label{sssec:equation-eqn-buy-with-fee-polynomial-inequality-derivation}

Assume $p_{ask} > p_m$.

\begin{equation*}
	p_{ask} > p_m
\end{equation*}

Substitute Equation~\ref{eqn:buy-with-fee-p_m}.

\begin{equation*}
	p_{ask} > \frac{q_i}{f^{\dagger} \cdot b_i}
\end{equation*}

Multiply both sides by $f^{\dagger} \cdot b_i$, which is nonnegative.

\begin{equation*}
	p_{ask} \cdot f^{\dagger} \cdot b_i > q_i
\end{equation*}

Substitute $t = (f^{\dagger} \cdot p_{ask} \cdot b_i)$ from
Equation~\ref{eqn:buy-with-fee-polynomial}.

\begin{equation*}
	t > q_i
\end{equation*}

Subtract $t$ from both sides.

\begin{equation*}
	0 > q_i - t
\end{equation*}

Rearrange.

\begin{equation*}
	q_i - t < 0
\end{equation*}

Multiply both sides by $q_i$.

\begin{equation*}
	q_i^2 - q_i \cdot t < 0
\end{equation*}

Substitute $C$ from Equation~\ref{eqn:buy-with-fee-polynomial}.

\begin{equation*}
	C < 0
\end{equation*}

Substitute $F(0) = C$ from Equation~\ref{eqn:buy-with-fee-polynomial}.

\begin{equation*}
	F(0) < 0
\end{equation*}
