\subsubsection{Equation~\ref{eqn:buy-with-fee-algorithm} derivation}%
\label{sssec:equation-eqn-buy-with-fee-algorithm-derivation}

Consider the update step from Equation~\ref{eqn:buy-with-fee-convergence}.

\begin{equation*}
	q_{n + 1} = \frac{q_n^2 + q_i \cdot t - q_i^2}
	{2 \cdot q_n + 2 \cdot q_i - f \cdot t}
\end{equation*}

Consider the first iteration where $q_n = q_0$ from Equation~\ref{eqn:buy-with-fee-q_0}.

\begin{equation*}
	q_1 = \frac{q_0^2 + q_i \cdot t - q_i^2}{2 \cdot q_0 + 2 \cdot q_i - f \cdot t}
\end{equation*}

Substitute $q_0$ from Equation~\ref{eqn:buy-with-fee-q_0}.

\begin{equation*}
	q_1 = \frac{{\left(\dfrac{f \cdot t}{2}\right)}^2 + q_i \cdot t - q_i^2}
	{2 \cdot \left(\dfrac{f \cdot t}{2}\right) + 2 \cdot q_i - f \cdot t}
\end{equation*}

Expand the denominator.

\begin{equation*}
	q_1 = \frac{{\left(\dfrac{f \cdot t}{2}\right)}^2 + q_i \cdot t - q_i^2}
	{f \cdot t + 2 \cdot q_i - f \cdot t}
\end{equation*}

Cancel like terms in the denominator.

\begin{equation*}
	q_1 = \frac{{\left(\dfrac{f \cdot t}{2}\right)}^2 + q_i \cdot t - q_i^2}
	{2 \cdot q_i}
\end{equation*}

Let $a = f \cdot t$, a recurring term across iterations, as well as $c = 2 \cdot q_i$.
Consider the recurring numerator term $b = q_i \cdot t - q_i^2$, which can be cached if
$b > 0$.

\begin{equation*}
	b = q_i \cdot t - q_i^2 \stackrel{?}{>} 0
\end{equation*}

Add $q_i^2$ to both sides.

\begin{equation*}
	q_i \cdot t \stackrel{?}{>} q_i^2
\end{equation*}

Per Equation~\ref{eqn:buy-with-fee-t-gt-q_i}, $t > q_i$ so long as $p_{ask} > p_{m}$, a
prerequisite for convergence. Thus the left side is always greater than the right side
and the inequality holds. Hence $q_1$ can be rewritten with recurring terms.

\begin{equation*}
	q_1 = \frac{{\left(\dfrac{a}{2}\right)}^2 + b}{c}
\end{equation*}

Similarly, $q_{n + 1}$ can be rewritten with recurring terms.

\begin{equation*}
	q_{n + 1} = \frac{q_n^2 + b}{2 \cdot q_n + c - a}
\end{equation*}
