\subsubsection{Equation~\ref{eqn:buy-with-fee-algorithm} derivation}%
\label{sssec:equation-eqn-buy-with-fee-algorithm-derivation}

Consider the update step from Equation~\ref{eqn:buy-with-fee-convergence}.

\begin{equation*}
	q_{s, n + 1} = \frac{q_{s, n}^2 + q_i \cdot t - q_i^2}
	{2 \cdot q_{s, n} + 2 \cdot q_i - f \cdot t}
\end{equation*}

Consider the first iteration where $q_{s, n} = q_{s, 0}$ from
Equation~\ref{eqn:buy-with-fee-q_s_0-eq-f_t}.

\begin{equation*}
	q_{s, 1} = \frac{q_{s, 0}^2 + q_i \cdot t - q_i^2}
	{2 \cdot q_{s, 0} + 2 \cdot q_i - f \cdot t}
\end{equation*}

Substitute $q_{s, 0} = f \cdot t$ from Equation~\ref{eqn:buy-with-fee-q_s_0-eq-f_t}.

\begin{equation*}
	q_{s, 1} = \frac{{(f \cdot t)}^2 + q_i \cdot t - q_i^2}
	{2 \cdot (f \cdot t) + 2 \cdot q_i - f \cdot t}
\end{equation*}

Combine like terms in the denominator.

\begin{equation*}
	q_{s, 1} = \frac{{(f \cdot t)}^2 + q_i \cdot t - q_i^2}{f \cdot t + 2 \cdot q_i}
\end{equation*}

Define the recurring terms $a = f \cdot t$, $b = q_i \cdot (t - q_i)$, and
$c = 2 \cdot q_i$.

\begin{equation*}
	q_{s, 1} = \frac{a^2 + b}{a + c}
\end{equation*}

Use recurring terms for the update step.

\begin{equation*}
	q_{s, n + 1} = \frac{q_{s, n}^2 + b}{2 \cdot q_{s, n} + c - a}
\end{equation*}

Note that in practice caching the recurring terms requires $b \geq 0$ to prevent
numerical underflow.

\begin{equation*}
	b = q_i \cdot (t - q_i) \stackrel{?}{>} 0
\end{equation*}

Divide both sides by $q_i$, which is nonnegative.

\begin{equation*}
	t - q_i \stackrel{?}{>} 0
\end{equation*}

Subtract $q_i$ from both sides.

\begin{equation*}
	t \stackrel{?}{>} q_i
\end{equation*}

Per Equation~\ref{eqn:buy-with-fee-t-gt-q_i}, $t > q_i$ so long as $p_{ask} > p_{m}$, a
condition that is also required for $q_{s, 0}$ per
Equation~\ref{eqn:buy-with-fee-q_s_0-eq-f_t}. Thus $b$ can be cached without underflow.
