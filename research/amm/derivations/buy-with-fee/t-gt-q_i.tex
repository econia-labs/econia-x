\subsubsection{Equation~\ref{eqn:buy-with-fee-t-gt-q_i} derivation}%
\label{sssec:equation-eqn-buy-with-fee-t-gt-q-i-derivation}

Assume $p_{ask} > p_m$

\begin{equation*}
	p_{ask} > p_m
\end{equation*}

Substitute Equation~\ref{eqn:buy-with-fee-p_m}.

\begin{equation*}
	p_{ask} > \frac{q_i}{f^{\dagger} \cdot b_i}
\end{equation*}

Multiply both sides by $(f^{\dagger} \cdot b_i)$.

\begin{equation*}
	p_{ask} \cdot f^{\dagger} \cdot b_i > q_i
\end{equation*}

Substitute $t$ from Equation~\ref{eqn:buy-with-fee-polynomial}.

\begin{equation*}
	t > q_i
\end{equation*}
